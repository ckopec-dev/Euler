- Find all numbers that can be expressed as the sum of a prime and twice a square

Let me implement a solution in Ada:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_376 is
   type Boolean_Array is array (Positive range <>) of Boolean;
   
   -- Sieve of Eratosthenes to find primes
   procedure Sieve_Of_Eratosthenes(Primes : in out Boolean_Array) is
   begin
      Primes(1) := False;
      for I in 2 .. Primes'Last loop
         if Primes(I) then
            for J in I * I .. Primes'Last loop loop
               if J mod I = 0 then
                  Primes(J) := False;
               end if;
            end loop;
         end if;
      end loop;
   end Sieve_Of_Eratosthenes;
   
   -- Generate primes up to limit
   function Get_Primes(Limit : Positive) return Boolean_Array is
      Primes : Boolean_Array(1 .. Limit);
   begin
      for I in 1 .. Limit loop
         Primes(I) := True;
      end loop;
      
      Sieve_Of_Eratosthenes(Primes);
      return Primes;
   end Get_Primes;
   
   -- Check if a number can be expressed as prime + 2*square
   function Can_Be_Expressed(Num : Positive; Primes : Boolean_Array) return Boolean is
      Square : Positive := 1;
      Test_Value : Positive;
   begin
      loop
         Test_Value := Num - 2 * Square * Square;
         exit when Test_Value < 2;
         
         if Test_Value <= Primes'Last and then Primes(Test_Value) then
            return True;
         end if;
         
         Square := Square + 1;
      end loop;
      
      return False;
   end Can_Be_Expressed;
   
   -- Main solution
   Limit : constant Positive := 1000000;
   Primes : Boolean_Array := Get_Primes(Limit);
   Count : Natural := 0;
   
begin
   for I in 1 .. Limit loop
      if Can_Be_Expressed(I, Primes) then
         Count := Count + 1;
      end if;
   end loop;
   
   Put("Answer: ");
   Put(Count, 0);
   New_Line;
end Euler_376;
```
