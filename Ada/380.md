- 2×1 dominoes (which can be placed horizontally or vertically)
- 1×1 squares

This is a classic dynamic programming problem.

- State 0: no cells filled in current column (all empty)

- State 3: one cell filled (bottom)

with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_380 is
   type State_Type is range 0 .. 7;
   
   -- Number of ways to tile a 3×n rectangle
   function Count_Tilings(n : Integer) return Long_Long_Integer is
      dp : array(0 .. n, 0 .. 7) of Long_Long_Integer := (others => (others => 0));
      
      -- Base case: empty rectangle
      dp(0, 0) := 1;
      
   begin
      -- Fill DP table
      for i in 1 .. n loop
         -- State 0 means all cells in column i are empty
         dp(i, 0) := dp(i-1, 7);  -- Previous state is all filled (7)
         
         -- State 1: top cell filled in current column
         dp(i, 1) := dp(i-1, 6);  -- Previous state has middle and bottom filled
         
         -- State 2: middle cell filled
         dp(i, 2) := dp(i-1, 5);  -- Previous state has top and bottom filled
         
         -- State 3: bottom cell filled  
         dp(i, 3) := dp(i-1, 4);  -- Previous state has top and middle filled
         
         -- State 4: top and middle cells filled
         dp(i, 4) := dp(i-1, 3);  -- Previous state has bottom filled
         
         -- State 5: top and bottom cells filled

         -- State 6: middle and bottom cells filled

         -- State 7: all cells filled
         dp(i, 7) := dp(i-1, 0);  -- Previous state empty
      end loop;
      
      return dp(n, 0);
