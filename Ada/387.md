# Euler Problem 387 Solution in Ada

I'll solve Project Euler Problem 387, which asks for the sum of all Harshad numbers less than 10^14 that are also divisible by the sum of their digits.

## Problem Analysis

A Harshad number is a number that is divisible by the sum of its digits. For example:
- 18 is a Harshad number because 1+8=9 and 18%9=0
- A right-truncatable Harshad number is one where all truncations from the right are also Harshad numbers
- We need to find all such numbers less than 10^14

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;
with Ada.Long_Long_Integer_Text_IO; use Ada.Long_Long_Integer_Text_IO;

procedure Euler_387 is
   
   type Long_Long is new Long_Long_Integer;
   
   Max_Value : constant Long_Long := 10**14;
   
   -- Calculate sum of digits
   function Sum_Of_Digits(N : Long_Long) return Long_Long is
      S : Long_Long := 0;
      Temp : Long_Long := N;
   begin
      while Temp > 0 loop
         S := S + (Temp mod 10);
         Temp := Temp / 10;
      end loop;
      return S;
   end Sum_Of_Digits;
   
   -- Check if number is Harshad
   function Is_Harshad(N : Long_Long) return Boolean is
   begin
      return N mod Sum_Of_Digits(N) = 0;
   end Is_Harshad;
   
   -- Check if number is right-truncatable Harshad
   function Is_Right_Truncatable_Harshad(N : Long_Long) return Boolean is
      Temp : Long_Long := N;
   begin
      while Temp > 0 loop
         if not Is_Harshad(Temp) then
            return False;
         end if;
         Temp := Temp / 10;
      end loop;
      return True;
   end Is_Right_Truncatable_Harshad;
   
   -- Check if number is strong Harshad (divisible by digit sum and quotient is prime)
   function Is_Strong_Harshad(N : Long_Long) return Boolean is
      Digit_Sum : Long_Long := Sum_Of_Digits(N);
   begin
      if N mod Digit_Sum = 0 then
         -- Check if N/Digit_Sum is prime
         declare
            Quotient : Long_Long := N / Digit_Sum;
            Is_Prime : Boolean := True;
            I : Long_Long := 2;
         begin
            if Quotient < 2 then
               return False;
            end if;
            
            while I * I <= Quotient loop
               if Quotient mod I = 0 then
                  Is_Prime := False;
                  exit;
               end if;
               I := I + 1;
            end loop;
            
            return Is_Prime;
         end;
      else
         return False;
      end if;
   end Is_Strong_Harshad;
   
   -- Recursive function to generate right-truncatable Harshad numbers
   procedure Generate_Right_Truncatable_Harshad(Num : Long_Long; Sum : Long_Long; Total : in out Long_Long) is
      Digit_Sum : Long_Long := Sum_Of_Digits(Num);
      New_Num : Long_Long;
   begin
      -- If this number is Harshad and right-truncatable, check if it's strong
      if Is_Right_Truncatable_Harshad(Num) then
         -- If it's a strong Harshad number (divisible by digit sum and quotient is prime)
         if Is_Strong_Harshad(Num) then
            Total := Total + Num;
         end if;
         
         -- Continue generating with digits 0-9 appended
         for D in 0..9 loop
            New_Num := Num * 10 + D;
            if New_Num < Max_Value then
               -- Check if the new number is still Harshad
               declare
                  New_Sum : Long_Long := Sum + D;
               begin
                  if New_Num mod New_Sum = 0 then
                     Generate_Right_Truncatable_Harshad(New_Num, New_Sum, Total);
                  end if;
               end;
            end if;
         end loop;
      end if;
   end Generate_Right_Truncatable_Harshad;
   
   -- Main function to solve the problem
   function Solve return Long_Long is
      Result : Long_Long := 0;
      Temp : Long_Long := 0;
   begin
      -- Start with single digit Harshad numbers (1-9)
      for I in 1..9 loop
         if Is_Harshad(I) then
            Generate_Right_Truncatable_Harshad(I, I, Result);
         end if;
      end loop;
      
      return Result;
   end Solve;
   
begin
   Put_Line("Project Euler Problem 387 Solution:");
   Put_Line("Sum of all strong Harshad numbers less than 10^14:");
   Put_Line(Long_Long'Image(Solve));
end Euler_387;
