I'll solve Euler Problem 452, which asks for the sum of all positive integers n ≤ 10^15 such that φ(n) = φ(n+1) = φ(n+2), where φ is Euler's totient function.

with Ada.Text_IO;

procedure Euler_452 is
   type Large_Integer is range 0 .. 10**15;
   
   -- Function to compute Euler's totient function
   function Phi(n : Large_Integer) return Large_Integer is
      result : Large_Integer := n;
      i      : Large_Integer := 2;
   begin
      while i * i <= n loop
         if n mod i = 0 then
            result := result - result / i;
            while n mod i = 0 loop
               n := n / i;
            end loop;
         end if;
         i := i + 1;
      end loop;
      
      if n > 1 then
         result := result - result / n;
      end if;
      
      return result;
   end Phi;
   
   -- Function to check if a number is prime
   function Is_Prime(n : Large_Integer) return Boolean is
   begin
      if n <= 1 then
         return False;
      elsif n <= 3 then
         return True;
      elsif n mod 2 = 0 or n mod 3 = 0 then
         return False;
      else
         declare
            i : Large_Integer := 5;
         begin
            while i * i <= n loop
               if n mod i = 0 or n mod (i + 2) = 0 then
                  return False;
               end if;
               i := i + 6;
            end loop;
            return True;
         end;
      end if;
   end Is_Prime;
   
   -- Function to compute gcd of two numbers
   function GCD(a, b : Large_Integer) return Large_Integer is
   begin
      while b /= 0 loop
         declare
            temp : Large_Integer := b;
         begin
            b := a mod b;
            a := temp;
         end;
      end loop;
      return a;
   end GCD;

   function Has_Consecutive_Equal_Phi(n : Large_Integer) return Boolean is
   begin
      return Phi(n) = Phi(n + 1) and then Phi(n + 1) = Phi(n + 2);
