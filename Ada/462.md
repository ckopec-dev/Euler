procedure Euler_462 is
   
   -- Function to calculate number of ways to tile 3×n rectangle with 2×1 dominoes
   function Tiling_Count(n : Integer) return Long_Long_Integer is
      F : array(0..50) of Long_Long_Integer;
   begin

         return 1;
      elsif n = 1 then
         return 1;
      elsif n = 2 then
         return 5;
      else
         F(0) := 1;
         F(1) := 1;
         F(2) := 5;

            F(i) := F(i-1) + 5 * F(i-2) + F(i-3);
         end loop;

      end if;
   end Tiling_Count;
