# Euler Problem 480 Solution in Ada

Euler Problem 480 asks us to find the number of ways to arrange the letters of the word "EUROPEAN" such that no two vowels are adjacent.

Let me break down the approach:
1. Identify the vowels and consonants
2. Arrange consonants first (they create slots for vowels)
3. Place vowels in the available slots
4. Calculate total arrangements

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_480 is
   -- Function to calculate factorial
   function Fact(N : Integer) return Long_Long_Integer is
   begin
      if N <= 1 then
         return 1;
      else
         return N * Fact(N - 1);
      end if;
   end Fact;
   
   -- Function to calculate combinations C(N,K)
   function Comb(N, K : Integer) return Long_Long_Integer is
   begin
      if K > N or K < 0 then
         return 0;
      elsif K = 0 or K = N then
         return 1;
      else
         return Fact(N) / (Fact(K) * Fact(N - K));
      end if;
   end Comb;
   
   -- Function to calculate arrangements with repeated elements
   function Arrangements(Counts : array of Integer) return Long_Long_Integer is
      Total : Long_Long_Integer := Fact(Sum(Counts));
      Denom : Long_Long_Integer := 1;
      Sum_Count : Integer := 0;
   begin
      for I in Counts'Range loop
         Sum_Count := Sum_Count + Counts(I);
         if Counts(I) > 1 then
            Denom := Denom * Fact(Counts(I));
         end if;
      end loop;
      
      return Total / Denom;
   end Arrangements;
   
   -- Main calculation
   procedure Solve is
      -- Letters in "EUROPEAN":
      -- E: 3 occurrences (vowels)
      -- U: 1 occurrence (vowel)  
      -- R: 2 occurrences (consonants)
      -- O: 1 occurrence (vowel)
      -- P: 1 occurrence (consonant)
      -- A: 1 occurrence (vowel)
      -- N: 1 occurrence (consonant)
      
      Vowels : array(1..5) of Character := ('E', 'U', 'O', 'A', 'E');  -- 5 vowels total, 3 E's
      Consonants : array(1..4) of Character := ('R', 'R', 'P', 'N');   -- 4 consonants total, 2 R's
      
      -- Total letters: 9
      -- We need to arrange so no two vowels are adjacent
      
      -- Step 1: Arrange consonants first
      -- We have 4 consonants with 2 R's
      Consonant_Arrangements : Long_Long_Integer := Fact(4) / Fact(2);  -- 4!/2! = 12
      
      -- Step 2: Create slots for vowels among consonants
      -- When we arrange 4 consonants, we create 5 slots (before first, between each, after last)
      -- _C_C_C_C_
      
      -- Step 3: Choose 5 positions from 5 available slots for 5 vowels
      -- This is C(5,5) = 1 way to choose positions
      
      -- Step 4: Arrange the 5 vowels in these 5 positions
      -- Vowels are: E, U, O, A, E (2 E's)
      Vowel_Arrangements : Long_Long_Integer := Fact(5) / Fact(2);  -- 5!/2! = 60
      
      -- Step 5: Total arrangements = consonant arrangements * vowel arrangements
      Result : Long_Long_Integer := Consonant_Arrangements * Vowel_Arrangements;
      
   begin
      Put_Line("Euler Problem 480 Solution:");
      Put_Line("Number of valid arrangements: ");
      Put(Long_Long_Integer'Image(Result));
      New_Line;
   end Solve;
   
begin
   Solve;
end Euler_480;

   function Fact(N : Integer) return Long_Long_Integer is

         return 1;

      end if;
   end Fact;
   
   procedure Solve is

      -- Letters: E, U, R, O, P, E, A, N, E (9 letters total)
      -- Vowels: E, U, O, A, E (5 vowels, 3 E's)

   begin
