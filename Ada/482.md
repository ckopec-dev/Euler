# Euler Problem 482 Solution in Ada

I'll solve Euler Problem 482, which asks for the sum of all distinct values of a^b + b^a where a and b are integers between 1 and 100.

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;
with Ada.Numerics.Big_Numbers.Big_Integers; use Ada.Numerics.Big_Numbers.Big_Integers;

procedure Euler_482 is
   type Big_Integer is new Ada.Numerics.Big_Numbers.Big_Integers.Big_Integer;
   
   -- Function to compute a^b using big integers
   function Power(Base : Big_Integer; Exp : Natural) return Big_Integer is
      Result : Big_Integer := 1;
   begin
      for I in 1..Exp loop
         Result := Result * Base;
      end loop;
      return Result;
   end Power;
   
   -- Function to compute a^b + b^a
   function Sum_Power(Base1, Base2 : Natural) return Big_Integer is
      B1, B2 : Big_Integer;
   begin
      B1 := Big_Integer(Base1);
      B2 := Big_Integer(Base2);
      return Power(B1, Base2) + Power(B2, Base1);
   end Sum_Power;
   
   -- Set to store distinct values
   type Value_Set is array (1..10000) of Big_Integer;
   Values : Value_Set;
   Count : Natural := 0;
   
   -- Check if value already exists in set
   function Is_Distinct(Value : Big_Integer) return Boolean is
      I : Natural;
   begin
      for I in 1..Count loop
         if Values(I) = Value then
            return False;
         end if;
      end loop;
      return True;
   end Is_Distinct;
   
   -- Add value to set if distinct
   procedure Add_Value(Value : Big_Integer) is
   begin
      if Is_Distinct(Value) then
         Count := Count + 1;
         Values(Count) := Value;
      end if;
   end Add_Value;
   
   Total_Sum : Big_Integer := 0;
   I, J : Natural;
begin
   -- Iterate through all pairs (a,b) where 1 <= a < b <= 100
   for I in 1..100 loop
      for J in (I+1)..100 loop
         Add_Value(Sum_Power(I, J));
      end loop;
   end loop;
   
   -- Also consider cases where a = b (a^b + b^a = 2*a^a)
   for I in 1..100 loop
      Add_Value(2 * Power(Big_Integer(I), I));
   end loop;
   
   -- Sum all distinct values
   for I in 1..Count loop
      Total_Sum := Total_Sum + Values(I);
   end loop;
   
   Put_Line("Sum of all distinct values: ");
   Put(Total_Sum);
   New_Line;
   
end Euler_482;
```

```ada
with Ada.Numerics.Big_Numbers.Big_Integers; use Ada.Numerics.Big_Numbers.Big_Integers;

procedure Euler_482 is
   type Big_Integer is new Ada.Numerics.Big_Numbers.Big_Integers.Big_Integer;
   
   -- Function to compute a^b using big integers
   function Power(Base : Big_Integer; Exp : Natural) return Big_Integer is
      Result : Big_Integer := 1;
   begin
      for I in 1..Exp loop
         Result := Result * Base;
      end loop;
      return Result;
   end Power;
   
   -- Function to compute a^b + b^a
   function Sum_Power(Base1, Base2 : Natural) return Big_Integer is
      B1, B2 : Big_Integer;
   begin
      B1 := Big_Integer(Base1);
      B2 := Big_Integer(Base2);
      return Power(B1, Base2) + Power(B2, Base1);
   end Sum_Power;
   
   -- Array to store distinct values (overkill but safe)
   type Distinct_Values is array (1..10000) of Big_Integer;
   Values : Distinct_Values;
   Count : Natural := 0;
   
   -- Check if value already exists in set
   function Is_Distinct(Value : Big_Integer) return Boolean is
      I : Natural;
   begin
      for I in 1..Count loop
         if Values(I) = Value then
            return False;
         end if;
      end loop;
      return True;
   end Is_Distinct;
   
   -- Add value to set if distinct
   procedure Add_Value(Value : Big_Integer) is
   begin
      if Is_Distinct(Value) then
         Count := Count + 1;
         Values(Count) := Value;
      end if;
   end Add_Value;
   
   Total_Sum : Big_Integer := 0;
   I, J : Natural;
begin
   -- Generate all a^b + b^a for 1 <= a,b <= 100
   for I in 1..100 loop
      for J in 1..100 loop
         Add_Value(Sum_Power(I, J));
      end loop;
   end loop;
   
   -- Sum all distinct values
   for I in 1..Count loop
      Total_Sum := Total_Sum + Values(I);
   end loop;
   
   Put_Line("Sum of all distinct values: ");
   Put(Total_Sum);
   New_Line;
   
end Euler_482;
```
