with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_495 is
   
   -- Problem parameters
   N : constant := 90;
   
   -- Define large integer type for handling big numbers
   type Big_Integer is range 0..2**63-1;
   
   -- Function to compute the number of ways to tile a 3×n rectangle
   function Tiling_Count(N : Integer) return Big_Integer is
      -- For n = 0, there's one way (empty tiling)
      -- For n = 1, there are 3 ways (three horizontal dominoes)
      -- For n = 2, there are 11 ways
      -- We'll compute this using dynamic programming
      
      DP : array(0..N) of Big_Integer;
      
   begin
      if N = 0 then
         return 1;
      elsif N = 1 then
         return 3;
      elsif N = 2 then
         return 11;
      end if;
      
      -- Base cases
      DP(0) := 1;
      DP(1) := 3;
      DP(2) := 11;
      
      -- Dynamic programming recurrence relation
      for I in 3..N loop
         -- The recurrence is: DP[i] = 4*DP[i-1] - DP[i-2]
         -- This comes from the mathematical analysis of 3×n tiling with dominoes
         DP(I) := 4 * DP(I-1) - DP(I-2);
      end loop;
      
      return DP(N);
   end Tiling_Count;
   
begin
   -- Compute the answer for n = 90
   declare
      Result : Big_Integer := Tiling_Count(90);
   begin
      Put("The number of ways to tile a 3×90 rectangle with 2×1 dominoes is: ");
      Put(Result, Width => 0);
      New_Line;
   end;
   
end Euler_495;
