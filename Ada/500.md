# Euler Problem 500 Solution in Ada

## Problem Statement
The number 12 = 2² × 3¹ has 6 divisors. The divisors are: 1, 2, 3, 4, 6, 12.

Find the smallest positive integer n such that n has exactly 500 divisors (including 1 and n).

## Solution Approach
To solve this problem, I need to:
1. Find the prime factorization of numbers
2. Calculate the number of divisors using the formula: if n = p₁^a₁ × p₂^a₂ × ... × pₖ^aₖ, then number of divisors = (a₁+1)(a₂+1)...(aₖ+1)
3. Find the smallest number with exactly 500 divisors

## Ada Implementation

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;
with Ada.Long_Long_Integer_Text_IO; use Ada.Long_Long_Integer_Text_IO;

procedure Euler_500 is
   
   type Prime_Array is array (Positive range <>) of Natural;
   
   -- Function to check if a number is prime
   function Is_Prime(N : Natural) return Boolean is
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         for I in 3 .. Integer'Sqrt(N) loop
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;
   
   -- Generate first N primes
   function Get_Primes(Count : Natural) return Prime_Array is
      Primes : Prime_Array(1 .. Count);
      Index : Positive := 1;
      Current : Natural := 2;
   begin
      while Index <= Count loop
         if Is_Prime(Current) then
            Primes(Index) := Current;
            Index := Index + 1;
         end if;
         Current := Current + 1;
      end loop;
      return Primes;
   end Get_Primes;
   
   -- Calculate number of divisors from prime factorization
   function Count_Divisors(Powers : array of Natural) return Natural is
      Result : Natural := 1;
   begin
      for I in Powers'Range loop
         Result := Result * (Powers(I) + 1);
      end loop;
      return Result;
   end Count_Divisors;
   
   -- Calculate number from prime factorization
   function Calculate_Number(Primes : Prime_Array; Powers : array of Natural) 
      return Long_Long_Integer is
      Result : Long_Long_Integer := 1;
   begin
      for I in Powers'Range loop
         for J in 1 .. Powers(I) loop
            Result := Result * Long_Long_Integer(Primes(I));
         end loop;
      end loop;
      return Result;
   end Calculate_Number;
   
   -- Check if a number is a perfect power
   function Is_Perfect_Power(N : Natural) return Boolean is
   begin
      if N < 2 then
         return False;
      end if;
      
      for I in 2 .. Integer'Sqrt(N) loop
         declare
            Temp : Natural := N;
            Count : Natural := 0;
         begin
            while Temp mod I = 0 loop
               Temp := Temp / I;
               Count := Count + 1;
            end loop;
            
            if Temp = 1 then
               return True;
            end if;
         end;
      end loop;
      
      return False;
   end Is_Perfect_Power;
   
   -- Main solution function
   function Solve_500_Divisors return Long_Long_Integer is
      Primes : Prime_Array(1 .. 100);  -- Generate enough primes
      Min_Number : Long_Long_Integer := Long_Long_Integer'Last;
      
      -- Simple approach: try combinations of prime powers
      procedure Try_Combinations(Powers : in out array of Natural; 
                                Index : Positive; 
                                Current_Divisors : Natural) is
         Divisor_Limit : constant Natural := 500;
         
         begin
            if Current_Divisors = Divisor_Limit then
               -- Calculate result
               declare
                  Result : Long_Long_Integer := Calculate_Number(Primes, Powers);
               begin
                  if Result < Min_Number then
                     Min_Number := Result;
                  end if;
               end;
               return;
            end if;
            
            if Current_Divisors > Divisor_Limit or Index > Primes'Length then
               return;
            end if;
            
            -- Try different powers for current prime
            for Power in 0 .. 20 loop
               Powers(Index) := Power;
               declare
                  New_Divisors : Natural := Current_Divisors;
               begin
                  if Power > 0 then
                     New_Divisors := Current_Divisors * (Power + 1);
                  end if;
                  
                  if New_Divisors <= Divisor_Limit then
                     Try_Combinations(Powers, Index + 1, New_Divisors);
                  else
                     exit; -- No point trying higher powers
                  end if;
               end;
            end loop;
         end Try_Combinations;
         
   begin
      -- Generate first 100 primes
      Primes := Get_Primes(100);
      
      -- Special case: We need exactly 500 divisors
      -- This is equivalent to finding a way to express 500 as product of factors
      -- 500 = 2² × 5³ = 4 × 125 = 25 × 20 = ...
      
      -- The key insight: if n = p₁^a₁ × p₂^a₂ × ... × pₖ^aₖ
      -- then number of divisors = (a₁+1)(a₂+1)...(aₖ+1) = 500
      
      -- We need to find the prime factorization of 500 and then 
      -- distribute these factors among exponents
      
      declare
         Powers : array(1 .. 100) of Natural := (others => 0);
      begin
         Try_Combinations(Powers, 1, 1);
      end;
      
      return Min_Number;
   end Solve_500_Divisors;

begin
   Put_Line("Solving Euler Problem 500...");
   declare
      Result : Long_Long_Integer := Solve_500_Divisors;
   begin
      Put("The smallest positive integer with exactly 500 divisors is: ");
      Put(Result);
      New_Line;
   end;
end Euler_500;
```
