# Euler Problem 505 Solution in Ada

I'll solve Euler Problem 505, which asks for the sum of all distinct pseudo-Fortunate numbers for primes p < 10^6.

First, let me understand what we're looking for:
- For each prime p, find the smallest positive integer m such that p + m is prime
- This m is called a pseudo-Fortunate number for p
- Sum all distinct pseudo-Fortunate numbers

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;
with Ada.Containers.Vectors;

procedure Euler_505 is
   
   -- Constants
   Max_Prime : constant := 1000000;
   
   -- Prime sieve array
   type Boolean_Array is array (Natural range <>) of Boolean;
   Primes : Boolean_Array(0..Max_Prime);
   
   -- Vector to store primes
   package Prime_Vector is new Ada.Containers.Vectors(Natural, Natural);
   use Prime_Vector;
   
   -- Function to check if a number is prime
   function Is_Prime(N : Natural) return Boolean is
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         for I in 3..(N-1) loop
            if I * I > N then
               return True;
            end if;
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;
   
   -- Sieve of Eratosthenes to find all primes up to Max_Prime
   procedure Sieve_Of_Eratosthenes is
   begin
      -- Initialize array
      for I in 0..Max_Prime loop
         Primes(I) := True;
      end loop;
      
      Primes(0) := False;
      Primes(1) := False;
      
      for I in 2..(Natural'Sqrt(Max_Prime)) loop
         if Primes(I) then
            for J in I*I..Max_Prime loop
               if J mod I = 0 then
                  Primes(J) := False;
               end if;
            end loop;
         end if;
      end loop;
   end Sieve_Of_Eratosthenes;
   
   -- Find the smallest m such that p + m is prime
   function Pseudo_Fortunate(P : Natural) return Natural is
      M : Natural := 1;
   begin
      loop
         if Primes(P + M) then
            return M;
         end if;
         M := M + 1;
      end loop;
   end Pseudo_Fortunate;
   
   -- Vector to store distinct pseudo-Fortunate numbers
   type Pseudo_Fortunate_Vector is array (Natural range <>) of Natural;
   Fortunate_Numbers : Pseudo_Fortunate_Vector(1..Max_Prime);
   Fortunate_Count : Natural := 0;
   
   -- Check if a number already exists in our collection
   function Contains(Number : Natural) return Boolean is
   begin
      for I in 1..Fortunate_Count loop
         if Fortunate_Numbers(I) = Number then
            return True;
         end if;
      end loop;
      return False;
   end Contains;
   
   -- Add a number to our collection if not already present
   procedure Add_Fortunate(Number : Natural) is
   begin
      if not Contains(Number) then
         Fortunate_Count := Fortunate_Count + 1;
         Fortunate_Numbers(Fortunate_Count) := Number;
      end if;
   end Add_Fortunate;
   
   -- Main solution
begin
   -- Initialize sieve
   Sieve_Of_Eratosthenes;
   
   -- Find all primes and their pseudo-Fortunate numbers
   for P in 2..Max_Prime loop
      if Primes(P) then
         declare
            M : Natural := Pseudo_Fortunate(P);
         begin
            Add_Fortunate(M);
         end;
      end if;
   end loop;
   
   -- Calculate sum of distinct pseudo-Fortunate numbers
   declare
      Sum : Natural := 0;
   begin
      for I in 1..Fortunate_Count loop
         Sum := Sum + Fortunate_Numbers(I);
      end loop;
      
      Put("Sum of distinct pseudo-Fortunate numbers: ");
      Put(Sum, 0);
      New_Line;
   end;
   
end Euler_505;
```
